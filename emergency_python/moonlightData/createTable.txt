지오메트리 칼럼 추가

ALTER TABLE public.moonlight_hospital
ADD COLUMN geom geometry(Point, 4326);

지오메트리 데이터 채우기

UPDATE public.moonlight_hospital
SET geom = ST_SetSRID(ST_MakePoint(wgs84lon, wgs84lat), 4326)
WHERE wgs84lon IS NOT NULL AND wgs84lat IS NOT NULL;

공간 인덱싱

CREATE INDEX idx_moonlight_hospital_geom
ON public.moonlight_hospital
USING GIST (geom);
